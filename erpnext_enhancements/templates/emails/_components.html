{# The email component macros. Every piece of styled markup this app emails lives here, exactly once. Nothing else in the app may contain an email hex colour or an email ; docs/email-design-system.md is the guide. TWO RULES, both load-bearing: 1. **This file may not reference `frappe`.** No get_url, no _(), no escape_html. Translated strings and absolute URLs arrive as arguments. That buys three things: `{% import %}` works without `with context`; the bench-free test renders this file under a vanilla jinja2 Environment with no frappe stub at all; and a macro has no route to site state. 2. **frappe's Jinja env has NO autoescape.** Every macro therefore `|e`s its own text arguments, so a caller may hand it raw user input. URLs are emitted UNESCAPED on purpose — entity-escaping a get_url() breaks the link — so callers must pass a URL they trust; email_style.button() runs them through utils.url_safety first. Styling is inline and already correct at 320px. The @media block in _shell.html only improves it; nothing here may depend on that block. #} {%- set TONES = { "primary": {"fg": "#0077a8", "bg": "#eef7fb", "br": "#b8dced"}, "success": {"fg": "#17753f", "bg": "#edf7f1", "br": "#b6ddc6"}, "warning": {"fg": "#8a5a00", "bg": "#fdf6e8", "br": "#e8d5a8"}, "danger": {"fg": "#a02a2a", "bg": "#fcefef", "br": "#e8bcbc"}, "info": {"fg": "#0077a8", "bg": "#f2f9fd", "br": "#b8dced"} } -%} {%- set INK = "#00263E" -%} {%- set BODY_INK = "#1f2d38" -%} {%- set MUTED = "#5b6b78" -%} {%- set RULE = "#dbe4ea" -%} {%- set SURFACE = "#f2f9fd" -%} {%- set SANS = "Helvetica,Arial,sans-serif" -%} {%- set MONO = "ui-monospace,SFMono-Regular,Menlo,Consolas,monospace" -%} {%- set WRAP = "word-break:break-word;overflow-wrap:anywhere" -%} {# A sub-heading inside the body. #} {% macro h(text) -%}

{# Label/value facts. Percentage split so it stays usable unstacked at 320px;
   the @media block stacks it when it can. `rows` is [(label, value), ...]. #}
{% macro kv(rows) -%}

{%- endfor %}


		{{ label|e }}
		{{ (value if value not in (None, "") else "—")|e }}
	

{%- endmacro %}


{# A data table. Keep to four columns or fewer — five does not fit a phone even
   stacked; if you need more, link to the report instead.

   A cell may be a plain value, or a ``(url, label)`` pair to render as a link.
   The label is escaped; the URL, as everywhere here, is not. #}
{% macro table(headers, rows) -%}

{%- for head in headers %}
		
{%- endfor %}
	
{%- for row in rows %}
	
{%- for cell in row %}
{%- if cell is not string and cell is iterable and cell|length == 2 %}
		
{%- else %}
		
{%- endif %}
{%- endfor %}
	
{%- endfor %}


	
{{ head|e }}
{{ cell[1]|e }}{{ (cell if cell not in (None, "") else "—")|e }}

{%- endmacro %}
